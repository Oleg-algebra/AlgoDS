T(n) = | O(1),              n = 0
       | 2 T(n - 1) + O(1), n >= 1

T(n) <=  2 T(n - 1) + C <= 2 (2T(n - 2) + C) + C = 4T(n-2) + C(2 + 1)<=
    <= 8T(n-3) + C(4+ 2 + 1) = 2^3 T(n-3) + C * sum_{i=0}^{3-1} 2^i =
    ... <= 2^n T(0) + C * sum_{i=0}^{n-1} 2^i <= 2^n C + C * sum_{i=0}^{n-1} 2^i = O(2^n)

---------b---------
T(n) = | O(1),              n = 0
       | 2 T(n - 1) + O(n), n >= 1

T(n) <= 2 T(n - 1) + C n  <= 2^2 T(n - 2) + C (n + 2(n-1))  <=
    <= 2^k T(n - k) + C (n + 2(n-1) + ... + 2^{k-1}(n-k+1)) <=
    <= 2^n T(n - n) + C (n + 2(n-1) + ... + 2^{n-1}(n- n + 1) <=
    <= 2^n C + C sum_{i=0}^{n-1} 2^i (n-i) = 2^n C + C(2^{n+1} - n -2)  = O(2^n)

T(n) = | O(1),                n = 1
       | 2 T([n / 2]) + O(1), n >= 2

n = 2^m - > m = log(n)

T(n) <= 2 T([n / 2]) + C = 2 T(2^{m-1}) + C <=
        <= 2^2 T(2^{m-2}) + C(2+1) <=
        <= 2^m T(1) + C(2^{m-1} + ... + 2+ 1) <=
        <=  C( 2^m + 2^{m-1} + ... + 2 + 1) = C (2^{m+1} - 1) = C(2*n - 1) = O(n)

T(n) = | O(1),                n = 1
       | 2 T([n / 2]) + O(n), n >= 2

T(n) <= 2 T(2^{m-1}) + C 2^m <=  2^2 T(2^{m-2}) + C 2 * 2^{m-1} + C 2^m = 2^2 T(2^{m-2}) + C(2*2^m) <=

        ...
        <= 2^m T(2^{m-m}) + C(m*2^m) <= 2^m  * C + C(m*2^m) = O(m*2^m) = O(n logn)
